}
 
export interface VitalLog {
    id: string;
    date: string;
    type: 'vital';
    vitalType: 'bp' | 'glucose' | 'weight';
    note?: string;
    // BP: { systolic, diastolic }
    // Glucose: { value, timing }
    // Weight: { kg }
    value: any;
    classification?: string; // e.g. 'Normal', 'Elevated', 'Pre-diabetic'
    classificationColor?: 'success' | 'warning' | 'error' | 'info';
}
